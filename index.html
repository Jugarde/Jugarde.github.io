def afficher_plateau(plateau):
    """Affiche le plateau de jeu de manière visuelle."""
    print("\n")
    print(f" {plateau[0]} | {plateau[1]} | {plateau[2]} ")
    print("---+---+---")
    print(f" {plateau[3]} | {plateau[4]} | {plateau[5]} ")
    print("---+---+---")
    print(f" {plateau[6]} | {plateau[7]} | {plateau[8]} ")
    print("\n")

def verifier_victoire(plateau, joueur):
    """Vérifie si le joueur actuel a gagné."""
    combinaisons = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8), # Lignes
        (0, 3, 6), (1, 4, 7), (2, 5, 8), # Colonnes
        (0, 4, 8), (2, 4, 6)             # Diagonales
    ]
    
    for a, b, c in combinaisons:
        if plateau[a] == plateau[b] == plateau[c] == joueur:
            return True
    return False

def jouer():
    """Fonction principale du jeu."""
    plateau = [" " for _ in range(9)]
    joueur_actuel = "X"
    jeu_en_cours = True

    print("Bienvenue dans le jeu du Morpion !")
    print("1 | 2 | 3")
    print("4 | 5 | 6")
    print("7 | 8 | 9")

    while jeu_en_cours:
        afficher_plateau(plateau)
        try:
            choix = int(input(f"Joueur {joueur_actuel}, case (1-9) : ")) - 1
            if 0 <= choix <= 8 and plateau[choix] == " ":
                plateau[choix] = joueur_actuel
                if verifier_victoire(plateau, joueur_actuel):
                    afficher_plateau(plateau)
                    print(f"Bravo ! {joueur_actuel} a gagné !")
                    jeu_en_cours = False
                elif " " not in plateau:
                    print("Match nul !")
                    jeu_en_cours = False
                else:
                    joueur_actuel = "O" if joueur_actuel == "X" else "X"
            else:
                print("Case invalide ou prise.")
        except ValueError:
            print("Entrez un chiffre valide.")

if __name__ == "__main__":
    jouer()

Extrait du script Python utilisant Pandas pour le traitement de l'emploi du temps. 

import numpy as np
import pandas as pd

# Chargement du fichier CSV
donnees = pd.read_csv("calendrier-sae-15.csv", sep=";")
donnees.columns = donnees.columns.str.strip()

def calculer_duree(heure_debut, heure_fin):
    deb = pd.to_datetime(heure_debut, format="%H:%M:%S", errors='coerce')
    fin = pd.to_datetime(heure_fin, format="%H:%M:%S", errors='coerce')
    if pd.isnull(deb) or pd.isnull(fin):
        return 0
    return (fin - deb).total_seconds() / 3600

def extraire_type(ligne):
    grp = str(ligne.get("Groupe", "")).upper()
    if "COURS" in grp: return "CM"
    elif "TD" in grp or "TDA" in grp or "TDB" in grp: return "TD"
    elif "TP" in grp or "PT" in grp: return "TP"
    else: return "Autre"

def extraire_matiere(texte):
    return str(texte).split(";")[0].strip().split()[0]

def repartition_BUT1():
    # On garde uniquement les lignes du BUT1
    donnees_but1 = donnees[donnees["Description"].str.contains("1A", na=False)].copy()
    
    # Calcul des durées
    donnees_but1["Duree"] = donnees_but1.apply(
        lambda ligne: calculer_duree(ligne["HStart"], ligne["HEnd"]), axis=1)
    
    # Extraction du type et de la matière
    donnees_but1["Type"] = donnees_but1.apply(extraire_type, axis=1)
    donnees_but1["Matiere"] = donnees_but1["Summary"].apply(extraire_matiere)

    # Application des coefficients selon le type
    def ajuster_volume(ligne):
        if ligne["Type"] == "CM": return ligne["Duree"]
        elif ligne["Type"] == "TD": return ligne["Duree"] / 2
        elif ligne["Type"] == "TP": return ligne["Duree"] / 4
        else: return 0

    donnees_but1["Volume_corrige"] = donnees_but1.apply(ajuster_volume, axis=1)
    
    # Tableau croisé final
    tableau_final = donnees_but1.pivot_table(
        values="Volume_corrige", index="Matiere", columns="Type",
        aggfunc=np.sum, fill_value=0
    )
    return tableau_final
